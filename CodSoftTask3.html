# 1. Bar Chart: Survival Rate by Passenger Class
# ------------------------------------------------------------------------------

import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

sns.set_theme(style="whitegrid")
df = pd.read_csv(r"C:\Users\HP\Downloads\titanic.csv")

plt.figure(figsize=(6, 4))
if 'Pclass' in df.columns and 'Survived' in df.columns:
    sns.barplot(
        x='Pclass',
        y='Survived',
        data=df,
        errorbar=None,
        hue='Pclass',
        palette='Blues_d',
        legend=False
    )
    plt.title('1. Survival Rate by Ticket Class', fontsize=12, fontweight='bold')
    plt.xlabel('Passenger Class (Pclass)')
    plt.ylabel('Survival Probability')
plt.tight_layout()
plt.show()

plt.figure(figsize=(6, 4))
if 'Sex' in df.columns:
    gender_counts = df['Sex'].value_counts()
    plt.pie(

        labels=[str(x).capitalize() for x in gender_counts.index],
        autopct='%1.1f%%',

# 5. Line Chart: Average Fare Trend Across Ages
# ------------------------------------------------------------------------------
plt.figure(figsize=(6, 4))
if 'Age' in df.columns and 'Fare' in df.columns:
    age_fare = df.groupby('Age')['Fare'].mean().reset_index()
    plt.plot(age_fare['Age'], age_fare['Fare'], color='darkred', linewidth=2)
    plt.title('5. Average Fare Price Across Age Groups', fontsize=12, fontweight='bold')
